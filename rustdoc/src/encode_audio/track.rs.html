11pub(crate) struct Track {
12    encoder: codec::encoder::audio::Encoder,
13    filter: AudioFilter,
14    stream: usize,
15    codec: Codec,
16    samples: u64,
17    payload_packets: u64,
18}
19
20impl Track {
21    /// Add a fully configured output stream before the muxer writes its header.
22    pub fn new(
23        kind: Codec,
24        config: &Config,
25        source: &frame::Audio,
26        input_time_base: Rational,
27        metadata: &Dictionary<'_>,
28        output: &mut format::context::Output,
29    ) -> Result<Self> {
30        let name = match kind {
31            Codec::Opus => "libopus",
32            Codec::Flac => "flac",
33            Codec::Truehd => "truehd",
34            Codec::Dual => bail!("dual is a container mode, not an encoder"),
35        };
36        let codec = ffmpeg::encoder::find_by_name(name)
37            .with_context(|| format!("encoder {name} unavailable in linked FFmpeg build"))?;
38        let audio_codec = codec.audio()?;
39        let layout = if kind == Codec::Truehd {
40            truehd_layout(source.channel_layout(), config.layout, &audio_codec)?
41        } else {
42            output_layout(source.channel_layout(), config.layout)?
43        };
44        if kind == Codec::Opus {
45            ensure!(
46                (1..=8).contains(&layout.channels()),
47                "Opus supports at most 8 channels in mapping family 1"
48            );
49        }
50        if let Some(layouts) = audio_codec.channel_layouts() {
51            ensure!(
52                layouts.into_iter().any(|l| l == layout),
53                "{name} does not support requested channel layout {layout:?}"
54            );
55        }
56        let rate = if kind == Codec::Opus {
57            48_000
58        } else {
59            source.rate()
60        };
61        ensure!(
62            rate > 0 && rate <= i32::MAX as u32,
63            "invalid source sample rate"
64        );
65        if let Some(rates) = audio_codec.rates() {
66            ensure!(
67                rates.into_iter().any(|r| r == rate as i32),
68                "{name} does not support {rate} Hz; refusing to resample lossless output"
69            );
70        }
71        let formats: Vec<_> = audio_codec
72            .formats()
73            .context("encoder advertises no sample formats")?
74            .collect();
75        let preferred = match kind {
76            Codec::Opus => format::Sample::F32(format::sample::Type::Packed),
77            Codec::Truehd if source.format().bytes() <= 2 => {
78                format::Sample::I16(format::sample::Type::Planar)
79            }
80            Codec::Truehd => format::Sample::I32(format::sample::Type::Planar),
81            _ if source.format().bytes() <= 2 => format::Sample::I16(format::sample::Type::Packed),
82            _ => format::Sample::I32(format::sample::Type::Packed),
83        };
84        ensure!(
85            formats.contains(&preferred),
86            "{name} lacks required sample format {preferred:?}"
87        );
88        let mut encoder = codec::context::Context::new_with_codec(codec)
89            .encoder()
90            .audio()?;
91        encoder.set_rate(rate as i32);
92        encoder.set_channel_layout(layout);
93        #[cfg(legacy_channel_count)]
94        encoder.set_channels(layout.channels());
95        encoder.set_format(preferred);
96        encoder.set_time_base((1, rate as i32));
97        if output
98            .format()
99            .flags()
100            .contains(format::flag::Flags::GLOBAL_HEADER)
101        {
102            encoder.set_flags(codec::flag::Flags::GLOBAL_HEADER);
103        }
104        let mut options = Dictionary::new();
105        let bitrate = config.bitrate_per_channel as usize * source.channels() as usize * 1000;
106        match kind {
107            Codec::Opus => {
108                ensure!(
109                    (6_000..=512_000).contains(&bitrate),
110                    "total Opus bitrate {bitrate} is outside libopus's 6000..=512000 bit/s range"
111                );
112                encoder.set_bit_rate(bitrate);
113                options.set("vbr", "on");
114                options.set(
115                    "mapping_family",
116                    if layout.channels() > 2 { "1" } else { "0" },
117                );
118            }
119            Codec::Flac => options.set("compression_level", "8"),
120            // FFmpeg marks this encoder experimental. This opt-in is confined
121            // to TrueHD rather than weakening compliance for all encoders.
122            Codec::Truehd => {
123                options.set("strict", "experimental");
124                options.set("max_interval", "16");
125            }
126            Codec::Dual => unreachable!(),
127        }
128        let encoder = encoder
129            .open_as_with(codec, options)
130            .with_context(|| format!("opening {name} encoder"))?;
131        let mut stream = output.add_stream(codec)?;
132        let stream_index = stream.index();
133        stream.set_parameters(&encoder);
134        stream.set_time_base(encoder.time_base());
135        let mut tags = metadata.to_owned();
136        if config.codec == Codec::Dual || kind == Codec::Truehd {
137            let title = match kind {
138                Codec::Opus => format!("Opus ({}k)", bitrate / 1000),
139                Codec::Flac => "FLAC (Lossless)".to_owned(),
140                _ => "TrueHD (Lossless)".to_owned(),
141            };
142            tags.set("title", &title);
143        }
144        stream.set_metadata(tags);
145        let filter = AudioFilter::new(source, &encoder, input_time_base)?;
146        Ok(Self {
147            encoder,
148            filter,
149            stream: stream_index,
150            codec: kind,
151            samples: 0,
152            payload_packets: 0,
153        })
154    }
155
156    pub fn push(
157        &mut self,
158        frame: &frame::Audio,
159        output: &mut format::context::Output,
160    ) -> Result<()> {
161        self.filter.push(frame)?;
162        self.drain_filter(output, false)
163    }
164
165    /// Flush resampler/FIFO before sending encoder EOF so delayed tail samples
166    /// are included. After EOF, EAGAIN is treated as an unexpected state.
167    pub fn finish(&mut self, output: &mut format::context::Output) -> Result<()> {
168        self.filter.finish()?;
169        self.drain_filter(output, true)?;
170        if self.codec == Codec::Truehd {
171            // The encoder buffers a major-sync interval. Very short inputs
172            // can otherwise yield packets that decode to no samples on 6.x.
173            let minimum = self.encoder.frame_size() as u64 * 16;
174            ensure!(self.samples >= minimum,
175                "TrueHD input is too short: need at least {minimum} samples per channel for a complete major-sync interval");
176        }
177        self.encoder.send_eof().context("flushing encoder")?;
178        self.drain_packets(output, true)?;
179        ensure!(
180            self.payload_packets > 0,
181            "encoder produced no audio packets"
182        );
183        Ok(())
184    }
185
186    fn drain_filter(
187        &mut self,
188        output: &mut format::context::Output,
189        finishing: bool,
190    ) -> Result<()> {
191        loop {
192            let mut frame = frame::Audio::empty();
193            match self.filter.receive(&mut frame) {
194                Ok(()) => {
195                    self.samples += frame.samples() as u64;
196                    let filter_time_base = self.filter.time_base()?;
197                    let pts = frame
198                        .pts()
199                        .map(|p| p.rescale(filter_time_base, self.encoder.time_base()));
200                    frame.set_pts(pts);
201                    self.encoder
202                        .send_frame(&frame)
203                        .context("sending frame to encoder")?;
204                    self.drain_packets(output, false)?;
205                }
206                Err(ffmpeg::Error::Eof) => return Ok(()),
207                Err(e) if is_again(e) && !finishing => return Ok(()),
208                Err(e) => return Err(e).context("receiving filtered audio"),
209            }
210        }
211    }
212
213    fn drain_packets(
214        &mut self,
215        output: &mut format::context::Output,
216        finishing: bool,
217    ) -> Result<()> {
218        loop {
219            let mut packet = Packet::empty();
220            match self.encoder.receive_packet(&mut packet) {
221                Ok(()) => {
222                    if packet.size() > 0 {
223                        self.payload_packets += 1;
224                    }
225                    // Muxers may rewrite stream time bases when writing headers.
226                    let time_base = output
227                        .stream(self.stream)
228                        .context("missing output stream")?
229                        .time_base();
230                    packet.rescale_ts(self.encoder.time_base(), time_base);
231                    packet.set_stream(self.stream);
232                    packet.set_position(-1);
233                    crate::mux::write(&mut packet, output)
234                        .with_context(|| format!("muxing {:?} packet", self.codec))?;
235                }
236                Err(ffmpeg::Error::Eof) => return Ok(()),
237                Err(e) if is_again(e) && !finishing => return Ok(()),
238                Err(e) => return Err(e).context("receiving encoded packet"),
239            }
240        }
241    }
242}

244/// Use the linked encoder's advertised layouts. TrueHD commonly wants side
245/// channels while libopus wants back channels; never silently reduce channels.
246fn truehd_layout(
247    source: ChannelLayout,
248    policy: LayoutPolicy,
249    codec: &codec::Audio,
250) -> Result<ChannelLayout> {
251    let supported: Vec<_> = codec
252        .channel_layouts()
253        .context("TrueHD advertises no layouts")?
254        .collect();
255    if supported.contains(&source) {
256        return Ok(source);
257    }
258    if policy == LayoutPolicy::Compatible {
259        // Newer FFmpeg bindings store AVChannelLayout's union internally.
260        // Such constants cannot be used in Rust structural match patterns;
261        // the wrapper's PartialEq implementation compares them safely.
262        let equivalent = if source == ChannelLayout::_5POINT1_BACK {
263            Some(ChannelLayout::_5POINT1)
264        } else if source == ChannelLayout::_5POINT0_BACK {
265            Some(ChannelLayout::_5POINT0)
266        } else {
267            None
268        };
269        if let Some(layout) = equivalent.filter(|l| supported.contains(l)) {
270            return Ok(layout);
271        }
272    }
273    bail!(
274        "linked TrueHD encoder does not support source layout {source:?}; supported: {supported:?}"
275    )
276}

278pub(crate) fn is_again(error: ffmpeg::Error) -> bool {
279    matches!(error, ffmpeg::Error::Other { errno } if errno == ffmpeg::error::EAGAIN)
